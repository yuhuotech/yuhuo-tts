## 📡 API完整调用示例

### Python 调用

```python
import requests
import base64
import json
from pathlib import Path

class TTSApiClient:
    def __init__(self, api_url="http://localhost:8000"):
        self.api_url = api_url
    
    # 1. 基础合成
    def synthesize_basic(self, text: str):
        """基础合成 - CosyVoice2, SFT模式"""
        url = f"{self.api_url}/synthesize"
        
        payload = {
            "text": text,
            "model": "cosyvoice2",
            "mode": "sft",
            "speaker": "default",
            "output_format": "base64"
        }
        
        response = requests.post(url, json=payload)
        result = response.json()
        
        if result['status'] == 'success':
            # 保存音频
            audio_base64 = result['data']['audio']
            audio_bytes = base64.b64decode(audio_base64)
            with open('output.wav', 'wb') as f:
                f.write(audio_bytes)
            
            # 打印时间戳
            print("\n字级时间戳:")
            for alignment in result['data']['alignments']:
                print(f"  {alignment['char']}: {alignment['start']:.3f}s - "
                      f"{alignment['end']:.3f}s (信心: {alignment['confidence']:.2f})")
        
        return result
    
    # 2. Qwen3合成
    def synthesize_qwen3(self, text: str):
        """Qwen3-TTS合成"""
        url = f"{self.api_url}/synthesize"
        
        payload = {
            "text": text,
            "model": "qwen3",
            "output_format": "url",      # 使用URL返回
            "save_file": True
        }
        
        response = requests.post(url, json=payload)
        return response.json()
    
    # 3. 获取模型列表
    def get_models(self):
        """获取可用模型"""
        url = f"{self.api_url}/models"
        response = requests.get(url)
        return response.json()
    
    # 4. 上传音频用于语音克隆
    def upload_audio(self, file_path: str):
        """上传音频文件"""
        url = f"{self.api_url}/upload_audio"
        
        with open(file_path, 'rb') as f:
            files = {'file': f}
            response = requests.post(url, files=files)
        
        return response.json()

# 使用示例
if __name__ == "__main__":
    client = TTSApiClient()
    
    # 示例1: CosyVoice2合成
    print("=== CosyVoice2 合成 ===")
    result = client.synthesize_basic("你好，今天天气真好")
    
    # 示例2: Qwen3合成
    print("\n=== Qwen3-TTS 合成 ===")
    result = client.synthesize_qwen3("这是一个多语言测试")
    
    # 示例3: 获取模型信息
    print("\n=== 可用模型 ===")
    models = client.get_models()
    for model_name, info in models['data'].items():
        print(f"{model_name}: sample_rate={info['sample_rate']}, "
              f"speakers={info['speakers']}")
```

### JavaScript/Node.js 调用

```javascript
// client.js
const axios = require('axios');
const fs = require('fs');

const API_URL = 'http://localhost:8000';

class TTSClient {
  async synthesize(text, model = 'cosyvoice2') {
    try {
      const response = await axios.post(`${API_URL}/synthesize`, {
        text,
        model,
        output_format: 'base64'
      });
      
      const { audio, alignments } = response.data.data;
      
      // 保存音频
      const buffer = Buffer.from(audio, 'base64');
      fs.writeFileSync('output.wav', buffer);
      
      // 打印时间戳
      console.log('字级时间戳:');
      alignments.forEach(({ char, start, end, confidence }) => {
        console.log(`  ${char}: ${start.toFixed(3)}s - ${end.toFixed(3)}s (${(confidence*100).toFixed(0)}%)`);
      });
      
      return response.data;
    } catch (error) {
      console.error('API调用失败:', error.message);
    }
  }
}

// 使用
const client = new TTSClient();
client.synthesize('你好世界');
```


### 流式处理大文本

```python
# 分批处理长文本
def synthesize_long_text(text: str, batch_size: int = 50):
    """分批合成长文本"""
    sentences = text.split('。')
    all_alignments = []
    current_time = 0
    
    for sentence in sentences:
        result = client.synthesize_basic(sentence)
        alignments = result['data']['alignments']
        
        # 调整时间戳
        for alignment in alignments:
            alignment['start'] += current_time
            alignment['end'] += current_time
        
        all_alignments.extend(alignments)
        current_time += result['data']['duration']
    
    return all_alignments
```
